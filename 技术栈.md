async handleScanClick() {
  // 显示加载状态

// 2. Scanner
async scanProfile(username, credentials) {
  const items = new Set();
  let scrollCount = 0;
  const maxScrolls = 50;
  
  while (scrollCount < maxScrolls) {
    // 滚动页面
    window.scrollTo(0, document.body.scrollHeight);
    await this.delay(2000);
    
    // 查找媒体元素
    const mediaElements = document.querySelectorAll('article img, article video');
    
    for (const el of mediaElements) {
      const url = this.extractMediaUrl(el);
      if (url && !items.has(url)) {
        items.add({
          url,
          type: el.tagName.toLowerCase() === 'video' ? 'video' : 'image',
          thumbnail: el.src,
          filename: this.generateFilename(url)
        });
      }
    }
    
    scrollCount++;
    
    // 检查是否到底
    if (this.isAtBottom()) break;
  }
  
  return Array.from(items);
}

// 3. Session Manager
async createSession(data) {
  const session = {
    id: this.generateId(),
    username: data.username,
    profileUrl: data.profileUrl,
    pendingItems: data.mediaItems,
    downloadedItems: [],
    failedItems: [],
    status: 'pending',
    progress: { completed: 0, total: data.mediaItems.length },
    createdAt: Date.now(),
    credentials: data.credentials
  };
  
  await this.storage.set(`session:${session.id}`, session);
  return session;
}

javascript
// 1. Download Manager
async startDownload({ sessionId, mediaItems, settings }) {
  // 加载会话
  const session = await this.sessionManager.getSession(sessionId);
  
  // 更新状态
  await this.sessionManager.updateSession(sessionId, {
    status: 'downloading',
    startedAt: Date.now()
  });
  
  // 创建队列
  const queue = this.createQueue(mediaItems, settings);
  
  // 并发下载
  const workers = [];
  for (let i = 0; i < settings.concurrentDownloads; i++) {
    workers.push(this.processQueue(queue, sessionId, settings));
  }
  
  // 等待完成
  await Promise.all(workers);
  
  // 完成
  await this.handleComplete(sessionId);
}

// 2. Queue Processor
async processQueue(queue, sessionId, settings) {
  while (queue.length > 0) {
    const item = queue.shift();
    
    try {
      // 下载延迟
      await this.randomDelay(settings.delayMin, settings.delayMax);
      
      // 执行下载
      await this.downloadItem(item, sessionId, settings);
      
      // 更新进度
      await this.updateProgress(sessionId, item);
      
      // 发送进度消息
      this.notifyProgress(sessionId);
      
    } catch (error) {
      // 重试逻辑
      if (item.retries < settings.maxRetries) {
        item.retries++;
        queue.push(item);
      } else {
        // 标记失败
        await this.markFailed(sessionId, item, error);
      }
    }
  }
}

// 3. Download Item
async downloadItem(item, sessionId, settings) {
  const filename = this.buildFilename(item, sessionId, settings);
  
  const downloadId = await chrome.downloads.download({
    url: item.url,
    filename: filename,
    saveAs: false
  });
  
  // 监听下载完成
  return new Promise((resolve, reject) => {
    const listener = (delta) => {
      if (delta.id === downloadId) {
        if (delta.state?.current === 'complete') {
          chrome.downloads.onChanged.removeListener(listener);
          resolve();
        } else if (delta.error) {
          chrome.downloads.onChanged.removeListener(listener);
          reject(new Error(delta.error.current));
        }
      }
    };
    
    chrome.downloads.onChanged.addListener(listener);
  });
}

// 4. Progress Update
async updateProgress(sessionId, item) {
  const session = await this.sessionManager.getSession(sessionId);
  
  session.downloadedItems.push(item);
  session.progress.completed++;
  session.progress.percentage = Math.round(
    (session.progress.completed / session.progress.total) * 100
  );
  
  await this.sessionManager.updateSession(sessionId, session);
}

// 5. Notify Progress
notifyProgress(sessionId) {
  chrome.runtime.sendMessage({
    type: 'DOWNLOAD_PROGRESS',
    payload: {
      sessionId,
      ...this.getProgress(sessionId)
    }
  });
}

class ProgressToast {
  constructor(sessionId) {

  create() {

    toast.className = 'progress-toast';

        </div>
        <div class="progress-text">
          <span class="progress-count">0 / 0</span>
          <span class="progress-percentage">0%</span>
        </div>
      </div>
    `;
    
    document.body.appendChild(toast);
    return toast;
  }
  
  show() {
    this.element.classList.add('show');
    this.startListening();
  }
  
  startListening() {
    chrome.runtime.onMessage.addListener((message) => {
      if (message.type === 'DOWNLOAD_PROGRESS' && 
          message.payload.sessionId === this.sessionId) {
        this.update(message.payload);
      }
    });
  }

// src/background/download-manager.js

class DownloadManager {
  async processQueue(queue, concurrency) {
    const running = new Set();
    const results = [];
    
    for (const item of queue) {
      // 创建下载任务
      const task = this.downloadItem(item)
        .then(result => {
          running.delete(task);
          results.push(result);
        })
        .catch(error => {
          running.delete(task);
          this.handleError(item, error);
        });
      
      running.add(task);
      
      // 等待并发数降低
      if (running.size >= concurrency) {
        await Promise.race(running);
      }
    }
    
    // 等待所有完成
    await Promise.all(running);
    
    return results;
  }
  
  async downloadItem(item) {
    // 添加随机延迟
    await this.randomDelay();
    
    // 执行下载
    const downloadId = await chrome.downloads.download({
      url: item.url,
      filename: this.buildFilename(item)
    });
    
    // 等待完成
    return this.waitForCompletion(downloadId);
  }
  
  randomDelay() {
    const min = this.settings.delayMin;
    const max = this.settings.delayMax;
    const delay = Math.random() * (max - min) + min;
    return new Promise(resolve => setTimeout(resolve, delay));
  }
}

class NetworkMonitor {
  constructor() {
    this.isOnline = navigator.onLine;
    this.retryQueue = [];
    this.setupListeners();
  }
  
  setupListeners() {
    window.addEventListener('online', () => {
      this.isOnline = true;
      this.processRetryQueue();
    });
    
    window.addEventListener('offline', () => {
      this.isOnline = false;
      this.pauseDownloads();
    });
  }
  
  async withRetry(fn, maxRetries = 3) {
    let lastError;
    
    for (let i = 0; i < maxRetries; i++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error;
        
        // 检查是否是网络错误
        if (this.isNetworkError(error)) {
          if (!this.isOnline) {
            // 添加到重试队列
            this.retryQueue.push({ fn, retriesLeft: maxRetries - i - 1 });
            throw error;
          }
          
          // 指数退避
          await this.exponentialBackoff(i);
        } else {
          // 非网络错误，直接抛出
          throw error;
        }
      }
    }
    
    throw lastError;
  }
  
  exponentialBackoff(attempt) {
    const delay = Math.min(1000 * Math.pow(2, attempt), 30000);
    return new Promise(resolve => setTimeout(resolve, delay));
  }
  
  isNetworkError(error) {
    return error.message.includes('network') ||
           error.message.includes('fetch') ||
           error.message.includes('timeout');
  }
}

class VirtualScroller {
  constructor(items, container, itemHeight) {
    this.items = items;
    this.container = container;
    this.itemHeight = itemHeight;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;
    
    this.setupScroll();
    this.render();
  }
  
  setupScroll() {
    this.container.addEventListener('scroll', () => {
      const scrollTop = this.container.scrollTop;
      const newStartIndex = Math.floor(scrollTop / this.itemHeight);
      
      if (newStartIndex !== this.startIndex) {
        this.startIndex = newStartIndex;
        this.render();
      }
    });
  }
  
  render() {
    const endIndex = Math.min(
      this.startIndex + this.visibleCount + 5,
      this.items.length
    );
    
    const visibleItems = this.items.slice(this.startIndex, endIndex);
    
    this.container.innerHTML = visibleItems
      .map((item, i) => this.createItem(item, this.startIndex + i))
      .join('');
    
    // 设置容器高度
    this.container.style.height = `${this.items.length * this.itemHeight}px`;
  }
}

function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}

// 节流
function throttle(func, limit) {
  let inThrottle;
  return function(...args) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

  async encrypt(data) {
    const encoder = new TextEncoder();
    const dataBuffer = encoder.encode(JSON.stringify(data));
    
    const key = await crypto.subtle.generateKey(
      { name: 'AES-GCM', length: 256 },
      true,
      ['encrypt', 'decrypt']
    );
    
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encrypted = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      key,
      dataBuffer
    );
    
    return { encrypted: Array.from(new Uint8Array(encrypted)), iv: Array.from(iv) };
  }
}

class RateLimiter {
  constructor(maxRequests, timeWindow) {
    this.maxRequests = maxRequests;
    this.timeWindow = timeWindow;
    this.requests = [];
  }
  
  async checkLimit() {
    const now = Date.now();
    
    // 清除过期请求
    this.requests = this.requests.filter(
      time => now - time < this.timeWindow
    );
    
    // 检查是否超限
    if (this.requests.length >= this.maxRequests) {
      const oldestRequest = this.requests[0];
      const waitTime = this.timeWindow - (now - oldestRequest);
      
      throw new Error(`Rate limit exceeded. Wait ${waitTime}ms`);
    }
    
    // 记录请求
    this.requests.push(now);
  }
}

const limiter = new RateLimiter(10, 60000); // 每分钟10次
